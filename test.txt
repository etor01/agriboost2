// ── Main Component ─────────────────────────────────────────────────
export default function MarketplaceClient() {
  const [dbListings, setDbListings] = useState<Listing[]>([]);
  const [loading, setLoading] = useState(true);

  // Original state declarations
  const [search, setSearch] = useState("");
  const [category, setCategory] = useState("All");
  const [region, setRegion] = useState("All Regions");
  const [sort, setSort] = useState("featured");
  const [cart, setCart] = useState<CartItem[]>([]);
  const [cartOpen, setCartOpen] = useState(false);
  const [page, setPage] = useState(1);
  const PER_PAGE = 24;

  useEffect(() => {
    async function fetchListings() {
      setLoading(true);

      // 1. Notice the new fields added to the .select() query
      const { data, error } = await supabase
        .from("listings")
        .select(
          `
          id,
          name,
          quantity,
          unit_type,
          price_per_unit,
          status,
          image_urls,
          rating,
          reviews,
          is_featured,
          discount,
          delivery_available,
          profiles ( location, verified_status ),
          crops ( crop_type, estimated_harvest_date )
        `,
        )
        .eq("status", "active");

      if (error) {
        console.error("Error fetching listings:", error);
      } else if (data) {
        // 2. Map the database columns directly to the frontend state
        const formattedListings: Listing[] = data.map((item: any) => ({
          id: item.id,
          title: item.name,
          category: item.crops?.crop_type || "General",
          price: Number(item.price_per_unit),
          unit: item.unit_type,
          quantity_available: Number(item.quantity),
          location: item.profiles?.location || "Unknown Location",
          is_verified: item.profiles?.verified_status || false,
          harvest_date:
            item.crops?.estimated_harvest_date || new Date().toISOString(),

          // These are now pulling live from Supabase!
          image_urls: item.image_urls || [
            "https://placehold.co/600x400/e2e8f0/1e293b?text=No+Image",
          ],
          rating: Number(item.rating) || 0,
          reviews: Number(item.reviews) || 0,
          is_featured: item.is_featured || false,
          discount: Number(item.discount) || 0,
          delivery_available: item.delivery_available ?? false,
        }));

        setDbListings(formattedListings);
      }
      setLoading(false);
    }

    fetchListings();
  }, []);
  const cartCount = cart.reduce((s, i) => s + i.qty, 0);
  const cartIds = new Set(cart.map((i) => i.listing.id));

  const addToCart = (listing: Listing) => {
    setCart((prev) => {
      const existing = prev.find((i) => i.listing.id === listing.id);
      if (existing)
        return prev.map((i) =>
          i.listing.id === listing.id ? { ...i, qty: i.qty + 1 } : i,
        );
      return [...prev, { listing, qty: 1 }];
    });
  };

  const removeFromCart = (id: string) =>
    setCart((prev) => prev.filter((i) => i.listing.id !== id));
  const changeQty = (id: string, qty: number) =>
    setCart((prev) =>
      prev.map((i) => (i.listing.id === id ? { ...i, qty } : i)),
    );

  // Switch out DUMMY_LISTINGS for dbListings here
  const filtered = useMemo(() => {
    let items = dbListings.filter((l) => {
      const q = search.toLowerCase();
      const matchSearch =
        !q ||
        l.title.toLowerCase().includes(q) ||
        l.location.toLowerCase().includes(q) ||
        l.category.toLowerCase().includes(q);
      const matchCat = category === "All" || l.category === category;
      const matchRegion = region === "All Regions" || l.location === region;
      return matchSearch && matchCat && matchRegion;
    });

    if (sort === "price_asc")
      items = [...items].sort((a, b) => a.price - b.price);
    else if (sort === "price_desc")
      items = [...items].sort((a, b) => b.price - a.price);
    else if (sort === "rating")
      items = [...items].sort((a, b) => b.rating - a.rating);
    else if (sort === "featured")
      items = [...items].sort(
        (a, b) => (b.is_featured ? 1 : 0) - (a.is_featured ? 1 : 0),
      );
    else if (sort === "newest")
      items = [...items].sort((a, b) =>
        b.harvest_date.localeCompare(a.harvest_date),
      );

    return items;
  }, [dbListings, search, category, region, sort]);

  const paginated = filtered.slice(0, page * PER_PAGE);
  const hasMore = paginated.length < filtered.length;

  if (loading) {
    return (
      <div className="flex h-screen w-full items-center justify-center bg-gray-50">
        <div className="text-emerald-600 font-bold text-xl animate-pulse">
          Loading Marketplace Data...
        </div>
      </div>
    );
  }

  return  <>
      {/* Cart Drawer */}
      {cartOpen && (
        <CartDrawer cart={cart} onClose={() => setCartOpen(false)} onRemove={removeFromCart} onQtyChange={changeQty} />
      )}

      {/* Sticky Cart Button */}
      {cartCount > 0 && (
        <button
          onClick={() => setCartOpen(true)}
          className="fixed bottom-6 right-6 z-40 bg-emerald-600 text-white px-5 py-3.5 rounded-2xl shadow-2xl hover:bg-emerald-700 transition-all active:scale-95 flex items-center gap-2.5 font-semibold"
        >
          <CartIcon className="h-5 w-5" />
          <span>Cart</span>
          <span className="bg-white text-emerald-700 text-xs font-extrabold w-5 h-5 rounded-full flex items-center justify-center">{cartCount}</span>
        </button>
      )}

      {/* Stats Bar */}
      <div className="bg-white/70 backdrop-blur-sm border-b border-white/50 py-3 px-4">
        <div className="max-w-7xl mx-auto flex flex-wrap items-center justify-between gap-3 text-sm">
          <div className="flex items-center gap-6">
            <span className="text-gray-600">🌾 <strong className="text-gray-900">{DUMMY_LISTINGS.length}</strong> products listed</span>
            <span className="text-gray-600">📍 <strong className="text-gray-900">10</strong> regions covered</span>
            <span className="text-gray-600">✅ <strong className="text-gray-900">{DUMMY_LISTINGS.filter(l => l.is_verified).length}</strong> verified farmers</span>
          </div>
          <button onClick={() => setCartOpen(true)} className="flex items-center gap-2 text-emerald-700 font-semibold hover:text-emerald-800">
            <CartIcon className="h-4 w-4" />
            Cart ({cartCount})
          </button>
        </div>
      </div>

      {/* Category Pills */}
      <div className="bg-white/60 backdrop-blur-sm border-b border-white/40 overflow-x-auto">
        <div className="max-w-7xl mx-auto px-4 py-3 flex gap-2 min-w-max">
          {ALL_CATEGORIES.map(cat => (
            <button
              key={cat}
              onClick={() => { setCategory(cat); setPage(1); }}
              className={`px-4 py-1.5 rounded-full text-sm font-medium whitespace-nowrap transition-all ${
                category === cat
                  ? "bg-emerald-600 text-white shadow-md"
                  : "bg-white text-gray-600 border border-gray-200 hover:border-emerald-400 hover:text-emerald-600"
              }`}
            >
              {cat}
            </button>
          ))}
        </div>
      </div>

      {/* Search & Filter Bar */}
      <div className="max-w-7xl mx-auto px-4 py-5">
        <div className="bg-white/80 backdrop-blur-sm rounded-2xl p-4 shadow-sm border border-white/60 flex flex-col md:flex-row gap-3">
          <div className="relative flex-grow">
            <SearchIcon className="absolute left-3.5 top-1/2 -translate-y-1/2 h-4.5 w-4.5 text-gray-400 h-5 w-5" />
            <input
              type="text"
              value={search}
              onChange={e => { setSearch(e.target.value); setPage(1); }}
              placeholder="Search maize, yams, tomatoes, spices..."
              className="w-full pl-10 pr-4 py-2.5 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-emerald-400 bg-white text-gray-800 placeholder-gray-400 text-sm"
            />
          </div>
          <div className="flex gap-3">
            <select
              value={region}
              onChange={e => { setRegion(e.target.value); setPage(1); }}
              className="px-3 py-2.5 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-emerald-400 bg-white text-gray-700 text-sm"
            >
              {REGIONS_FILTER.map(r => <option key={r}>{r}</option>)}
            </select>
            <select
              value={sort}
              onChange={e => setSort(e.target.value)}
              className="px-3 py-2.5 rounded-xl border border-gray-200 focus:outline-none focus:ring-2 focus:ring-emerald-400 bg-white text-gray-700 text-sm"
            >
              <option value="featured">⭐ Featured First</option>
              <option value="newest">🌱 Newest Harvest</option>
              <option value="price_asc">Price: Low → High</option>
              <option value="price_desc">Price: High → Low</option>
              <option value="rating">⭐ Highest Rated</option>
            </select>
          </div>
        </div>

        {/* Result count */}
        <div className="flex items-center gap-2 mt-3 mb-5 text-sm text-gray-600">
          <FilterIcon className="h-4 w-4" />
          <span>Showing <strong className="text-gray-900">{paginated.length}</strong> of <strong className="text-gray-900">{filtered.length}</strong> products</span>
          {(search || category !== "All" || region !== "All Regions") && (
            <button onClick={() => { setSearch(""); setCategory("All"); setRegion("All Regions"); setPage(1); }} className="ml-2 text-xs text-red-400 hover:text-red-600 underline">
              Clear filters
            </button>
          )}
        </div>

        {/* Grid */}
        {filtered.length === 0 ? (
          <div className="text-center py-24 bg-white/70 rounded-2xl border-2 border-dashed border-gray-200">
            <p className="text-4xl mb-3">🌾</p>
            <h3 className="text-xl font-semibold text-gray-800">No products found</h3>
            <p className="text-gray-500 mt-1">Try adjusting your search or filters.</p>
          </div>
        ) : (
          <>
            <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4 gap-5">
              {paginated.map(listing => (
                <ProductCard
                  key={listing.id}
                  listing={listing}
                  onAddToCart={addToCart}
                  inCart={cartIds.has(listing.id)}
                />
              ))}
            </div>
            {hasMore && (
              <div className="text-center mt-10">
                <button
                  onClick={() => setPage(p => p + 1)}
                  className="px-8 py-3 bg-white border-2 border-emerald-500 text-emerald-600 font-semibold rounded-xl hover:bg-emerald-50 transition-colors shadow-sm"
                >
                  Load More Products ({filtered.length - paginated.length} remaining)
                </button>
              </div>
            )}
          </>
        )}
      </div>
    </>;
}
